import React, { useState } from "react";
import { View, Text, Pressable } from "react-native";
export default function ConsolidacaoFastRefresh() {
 const [count, setCount] = useState(0);
 return <View style={{ flex: 1, justifyContent: "center", alignItems: "center", backgroundColor: "#F7F4EC", gap: 24 }}>
  <Text style={{ fontSize: 24, color: "#10162F" }}>Consolidacao refresh A</Text>
  <Text style={{ fontSize: 24, color: "#10162F" }}>Contador {count}</Text>
  <Pressable accessibilityRole="button" accessibilityLabel="Incrementar diagnostico" onPress={() => setCount((value) => value + 1)} style={{ padding: 24, backgroundColor: "#EFE7FF" }}><Text>Incrementar diagnostico</Text></Pressable>
 </View>;
}
